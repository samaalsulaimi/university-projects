// Function to load a BMP file and convert it to an OpenGL Texture
//http://stackoverflow.com/questions/12518111/how-to-load-a-bmp-on-glut-to-use-it-as-a-texture
GLuint LoadTexture(const char* filename, int width, int height)
{
    GLuint texture;
    unsigned char* data;
    FILE* file;

    //The following code will read in our RAW file
    fopen_s(&file, filename, "rb");

    if (file == NULL)
    {
        cout << "Unable to open the image file" << endl << "Program will exit :(" << endl;
        exit(0);
        return 0;
    }

    data = (unsigned char*)malloc(width * height * 3);
    fread(data, width * height * 3, 1, file);

    fclose(file);


    // reorder the image colors to RGB not BGR
    for (int i = 0; i < width * height; ++i)
    {
        int index = i * 3;
        unsigned char B, R;
        B = data[index];
        R = data[index + 2];

        data[index] = R;
        data[index + 2] = B;

    }

    glGenTextures(1, &texture);            //generate the texture with the loaded data
    glBindTexture(GL_TEXTURE_2D, texture); //bind the texture to it's array

    glTexEnvf(GL_TEXTURE_ENV, GL_TEXTURE_ENV_MODE, GL_REPLACE); //set texture environment parameters

    glTexParameterf(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR);
    glTexParameterf(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
    glTexParameterf(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_REPEAT);
    glTexParameterf(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_REPEAT);

    glTexImage2D(GL_TEXTURE_2D, 0, GL_RGB, width, height, 0, GL_RGB, GL_UNSIGNED_BYTE, data);

    free(data); //free the texture array

    if (glGetError() != GL_NO_ERROR)
        printf("GLError in genTexture()\n");

    return texture; //return whether it was successfull  
}

//Basic graphing functions
void drawCircle(float radius, float r, float g, float b, float alpha = 1.0f) {
    glColor4f(r, g, b, alpha);
    glBegin(GL_POLYGON);
    for (int i = 0; i < 360; i += 10) {
        float theta = i * 3.14159f / 180.0f;
        glVertex2f(cos(theta) * radius, sin(theta) * radius);
    }
    glEnd();
}
//Helper function to draw a triangle(for panda's nose) 
void drawTriangle(float width, float height, float r, float g, float b) {
    glColor3f(r, g, b);    // Set the triangle color
    glBegin(GL_TRIANGLES); // Start drawing a triangle
    glVertex2f(-width / 2.0f, height / 2.0f); // Top-left vertex
    glVertex2f(width / 2.0f, height / 2.0f);  // Top-right vertex
    glVertex2f(0.0f, -height / 2.0f);         // Bottom-center vertex
    glEnd();               // End drawing the triangle
}

// --- Draw an apple tree ---
void drawAppleTree(float x, float y, float scale) {
    glPushMatrix();               // Isolate tree transformations (Save matrix state)
    glTranslatef(x, y, 0);        // Move tree to designated coordinates
    glScalef(scale, scale, 1.0f); // Apply overall tree scaling

    // Draw tree trunk
    glColor3f(0.4f, 0.2f, 0.0f);  // Brown color
    glBegin(GL_QUADS);            // Trunk drawn as a rectangle
    glVertex2f(-0.3f, -2.5f);
    glVertex2f(0.3f, -2.5f);
    glVertex2f(0.3f, 0.0f);
    glVertex2f(-0.3f, 0.0f);
    glEnd();

    // Draw foliage (3 green circles) at different positions
    glPushMatrix(); glTranslatef(-0.7f, 0.2f, 0); drawCircle(1.0f, 0.0f, 0.5f, 0.0f); glPopMatrix();
    glPushMatrix(); glTranslatef(0.7f, 0.2f, 0); drawCircle(1.0f, 0.0f, 0.5f, 0.0f); glPopMatrix();
    glPushMatrix(); glTranslatef(0.0f, 1.0f, 0); drawCircle(1.1f, 0.0f, 0.6f, 0.0f); glPopMatrix();

    // Draw 5 red apples using an array of coordinates
    float apples[5][2] = { {0.0f, 1.5f}, {0.6f, 0.5f}, {-0.6f, 0.5f}, {0.5f, -0.1f}, {-0.5f, -0.1f} };
    for (int i = 0; i < 5; i++) {
        glPushMatrix();
        glTranslatef(apples[i][0], apples[i][1], 0); // Translate to each apple's position
        drawCircle(0.15f, 1.0f, 0.0f, 0.0f);          // Draw small red circle
        glPopMatrix();
    }
    glPopMatrix(); // Restore original matrix (undo translation and scaling)
}

// --- Draw a panda ---
void drawPanda(float size, bool isMother) {
    glPushMatrix();                     // Save matrix for panda context
    drawCircle(size, 1.0f, 1.0f, 1.0f); // Draw the body (White circle)

    // Draw the head
    glPushMatrix();
    glTranslatef(0, size * 0.85f, 0);           // Move head slightly above the body
    drawCircle(size * 0.65f, 1.0f, 1.0f, 1.0f); // Draw head (White circle)

    // Ears, Eyes, and Nose
    glPushMatrix(); glTranslatef(-size * 0.55f, size * 0.55f, 0); drawCircle(size * 0.25f, 0, 0, 0); glPopMatrix(); // Left ear (black)
    glPushMatrix(); glTranslatef(size * 0.55f, size * 0.55f, 0); drawCircle(size * 0.25f, 0, 0, 0); glPopMatrix();  // Right ear (black)
    glPushMatrix(); glTranslatef(-size * 0.25f, size * 0.15f, 0.1f); drawCircle(size * 0.15f, 0, 0, 0); // Left eye patch
    glTranslatef(size * 0.5f, 0, 0); drawCircle(size * 0.15f, 0, 0, 0); glPopMatrix();                   // Right eye patch
    glPushMatrix(); glTranslatef(0.0f, -size * 0.1f, 0.1f); drawTriangle(size * 0.22f, size * 0.15f, 0, 0, 0); glPopMatrix(); // Triangular nose
    glPopMatrix(); // End head context

    // Set hand size based on whether it's a mother or cub
    float handSizeFactor = isMother ? 0.35f : 0.25f;
    for (int i = 0; i < 2; i++) {
        float xSide = (i == 0) ? -1.0f : 1.0f; // Toggle between left and right side
        glPushMatrix(); glTranslatef(xSide * size * 0.95f, size * 0.1f, 0); drawCircle(size * handSizeFactor, 0, 0, 0); glPopMatrix(); // Draw hand
        glPushMatrix(); glTranslatef(xSide * size * 0.5f, -size * 0.95f, 0); drawCircle(size * 0.3f, 0, 0, 0); glPopMatrix();         // Draw foot
    }
    glPopMatrix(); // Restore matrix
}

// Draw text on screen
void drawText(float x, float y, const char* text, bool isWhite) {

    // Color
    if (isWhite)
        glColor3f(1.0f, 1.0f, 1.0f); // White
    else
        glColor3f(0.0f, 0.0f, 0.0f); // Black

    // Position
    glRasterPos2f(x, y);

    // Taxt
    while (*text)
        glutBitmapCharacter(GLUT_BITMAP_TIMES_ROMAN_24, *text++);
}

// Resets all object positions and flags when switching between story scenes
void resetSceneAnimations() {
    pandaWalk = -15.0f;
    rabbitMove = 0;
    rabbitHop = 0;
    pandaWalkX = -10.0f;
    momX = -25.0f;
    childX = 3.5f;
    heartsFinished = false;
    initialized = false;
    for (int i = 0; i < 3; i++) {
        myHearts[i].alpha = 1;
        myHearts[i].active = true;
    }
}
// Handles keyboard input for scene switching (1-5)
void keyboard(unsigned char key, int x, int y) {
    if (key == '1') { currentScene = 1; resetSceneAnimations(); }
    if (key == '2') { currentScene = 2; resetSceneAnimations(); }
    if (key == '3') { currentScene = 3; resetSceneAnimations(); }
    if (key == '4') { currentScene = 4; resetSceneAnimations(); }
    if (key == '5') { currentScene = 5; resetSceneAnimations(); }

    glutPostRedisplay();
}
// --- Basic glut functions ---

void display() {
    glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);
    glMatrixMode(GL_MODELVIEW);
    glLoadIdentity();
    if (currentScene == 1) drawScene1();
    else if (currentScene == 2) drawScene2();
    else if (currentScene == 3) drawScene3();
    else if (currentScene == 4) drawScene4();
    else if (currentScene == 5) drawScene5();

    glutSwapBuffers();
}
void init() {
    myTexture1 = LoadTexture(image1Path, 720, 510);

    if (myTexture1 == -1)
    {
        cout << "Error in loading the texture" << endl;
    }
    else
        cout << "The texture value is: " << myTexture1 << endl;
    myTexture2 = LoadTexture(image2Path, 1024, 725);

    glEnable(GL_TEXTURE_2D);
    //----------------------

    glClearColor(0.70f, 0.70f, 0.70f, 1.0f); // Set background color to black and opaque
    // Set 2D projection (IMPORTANT)
    glMatrixMode(GL_PROJECTION);
    glLoadIdentity();
    glOrtho(-20, 20, -15, 15, -1, 1);

    glMatrixMode(GL_MODELVIEW);

    // Enable transparency (for rain/hearts)
    glEnable(GL_BLEND);
    glBlendFunc(GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA);

    // Preparing rain drops
    srand(time(NULL));
    for (int i = 0; i < numDrops; i++) {
        RainDrop drop;
        drop.x = (rand() % 40) - 20.0f; // Between 20 and -20
        drop.y = (rand() % 30) - 15.0f; // Between 15 and -15
        drop.speed = 0.2f + (rand() % 10) / 50.0f; // Varying speed
        rainDrops.push_back(drop);
    }
}
void reshape(int w, int h) {
    glViewport(0, 0, w, h);
    glMatrixMode(GL_PROJECTION);
    glLoadIdentity();
    // Same coordinate system
    glOrtho(-20, 20, -15, 15, -1, 1);
    glMatrixMode(GL_MODELVIEW);
}
// Timer function executed every ~16ms (60 FPS)
void timer(int v) {
    timerCount += 0.05f;
    windSwing = sin(timerCount) * 4;
    // Moving panda mom
    pandaWalkX += 0.020f; // slow speed
    if (pandaWalkX > 22.0f) pandaWalkX = -22.0f; // Location changed
    moveSmallPanda += 0.02f;     // Increment counter for panda movement
    // Rabbit moves only before panda reaches the middle
    if (pandaWalk < 3) {
        rabbitHop = sin(timerCount * 6) * 0.4f;
        rabbitMove += 0.02f * rabbitDir;

        if (rabbitMove > 1.0f || rabbitMove < -1.0f)
            rabbitDir *= -1; // Change direction
    }
    else {
        rabbitHop = 0;
        rabbitMove = 0;
    }

    // Panda stops when speaking
    float pandaSpeed = (pandaWalk > 3 && pandaWalk < 6) ? 0.0f : 0.03f;
    pandaWalk += pandaSpeed;

    // Reset panda position after leaving screen
    if (pandaWalk > 22) pandaWalk = -22;


    // Moving rain drops
    for (auto& drop : rainDrops) {
        drop.y -= drop.speed; // drops down
        drop.x -= 0.02f;


        if (drop.y < -15.0f) {
            drop.y = 15.0f;
            drop.x = (rand() % 40) - 20.0f;
        }
    }



    if (momX < childX - 1.5f) {
        momX += 0.1f;
    }
    else {
        if (!initialized) {
            initHearts();
            initialized = true;
        }

        bool anyActive = false;

        for (int i = 0; i < 3; i++) {
            if (myHearts[i].alpha > 0.01f) {

                float pulse = 1.0f + sin(timerCount * 6 + i) * 0.1f;

                if (i == 0) {
                    myHearts[i].size = 0.5f * pulse;
                }
                else if (i == 1) {
                    myHearts[i].size = 0.8f * pulse;
                }
                else {
                    myHearts[i].size = 0.6f * pulse;
                }

                myHearts[i].alpha -= 0.01f;

                anyActive = true;
            }
            else {
                myHearts[i].alpha = 0;
                myHearts[i].active = false;
            }
        }

        if (!anyActive) {
            heartsFinished = true;
        }
    }


    glutPostRedisplay();// Request scene refresh
    glutTimerFunc(16, timer, 0);// Re-trigger timer every 16ms (~60 FPS)

}

int main(int argc, char** argv)
{

    glutInit(&argc, argv);               					     	 // Initialize GLUT
    glutInitDisplayMode(GLUT_RGBA | GLUT_DOUBLE | GLUT_DEPTH);     //  Specify the display Mode ñ RGB, RGBA or color                                                        //  Index, single or double Buffer
    glutInitWindowSize(1000, 800);                                    // Set the window's initial width & height
    glutInitWindowPosition(250, 50); 								 // Position the window's initial top-left corner
    glutCreateWindow("Full Project");    							 // Create a window with the given title
    init();
    glutDisplayFunc(display);
    glutReshapeFunc(reshape);
    glutKeyboardFunc(keyboard);
    glutTimerFunc(0, timer, 0);
    glutMainLoop();           		// Enter the infinitely event-processing loop

    return 0;
}
